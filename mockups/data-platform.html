-- Incremental load of orders, generated from the "Incremental (CDC)" pattern MERGE INTO ANALYTICS.FACT_ORDERS AS t USING ( SELECT o.order_id, o.customer_id, o.order_date, o.status, SUM(d.quantity * d.unit_price) AS amount FROM sales.orders o JOIN sales.order_details d ON d.order_id = o.order_id WHERE o.updated_at > '{{ last_run_at }}' GROUP BY 1, 2, 3, 4 ) AS s ON t.order_id = s.order_id WHEN MATCHED THEN UPDATE SET t.status = s.status, t.amount = s.amount WHEN NOT MATCHED THEN INSERT (order_id, customer_id, order_date, status, amount) VALUES (s.order_id, s.customer_id, s.order_date, s.status, s.amount);
